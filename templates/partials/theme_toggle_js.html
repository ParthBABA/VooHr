// Light/dark theme toggle. // // Extracted verbatim from the seven pages that carried an identical copy of // this handler (dashboard, conversation-workspace, risk-drift, meeting_tracker, // settings, sync, sync_room). The persisted key, the light/dark attribute // handling and the swallow-on-private-mode localStorage write are unchanged. (function() { var btn = document.getElementById('themeToggle'); if (!btn) return; btn.addEventListener('click', function() { var current = document.documentElement.getAttribute('data-theme') === 'light' ? 'light' : 'dark'; var next = current === 'light' ? 'dark' : 'light'; if (next === 'dark') { document.documentElement.removeAttribute('data-theme'); } else { document.documentElement.setAttribute('data-theme', 'light'); } try { localStorage.setItem('voovr-theme', next); } catch (e) {} }); })();
